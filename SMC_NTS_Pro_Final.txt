    bullAboveDown = not na(downNow) and not na(downPrev) and close > downNow + tlTouchTolAtr * atrSeed and close[1] <= downPrev + tlTouchTolAtr * atrSeed
    bullAboveUp = not na(upNow) and not na(upPrev) and close > upNow + tlTouchTolAtr * atrSeed and close[1] <= upPrev + tlTouchTolAtr * atrSeed
    bearBelowUp = not na(upNow) and not na(upPrev) and close < upNow - tlTouchTolAtr * atrSeed and close[1] >= upPrev - tlTouchTolAtr * atrSeed
    bearBelowDown = not na(downNow) and not na(downPrev) and close < downNow - tlTouchTolAtr * atrSeed and close[1] >= downPrev - tlTouchTolAtr * atrSeed
    trendlineBullBreak := (bullAboveDown or bullAboveUp) and barstate.isconfirmed
    trendlineBearBreak := (bearBelowUp or bearBelowDown) and barstate.isconfirmed
trendlineBuyZone = not na(pdRange) and pdRange > 0 and close <= pdLow + pdRange * tlDiscountEndPct / 100.0
trendlineSellZone = not na(pdRange) and pdRange > 0 and close >= pdLow + pdRange * tlPremiumStartPct / 100.0
